   119	def test_no_claim_type_equality_branching_in_package() -> None:
   120	    """Decision logic about claim type lives in the model, not in Python.
   121	
   122	    tools.py is exempt (it defines the enum in the input_schema).
   123	    pricing.py is exempt (it may map claim_type to per-queue cost weights).
   124	    """

   126	    offenders: list[str] = []
   127	
   128	    for py_file in PKG.rglob("*.py"):

   130	            continue
   131	        tree = _parse(py_file)
   132	        for node in ast.walk(tree):
   133	            if not isinstance(node, ast.Compare):
   134	                continue
   135	            if not any(isinstance(op, ast.Eq) for op in node.ops):
   136	                continue

   138	            left_name = (
   139	                left.attr if isinstance(left, ast.Attribute)
   140	                else left.id if isinstance(left, ast.Name)
   141	                else None
   142	            )
   143	            if left_name != "claim_type":
   144	                continue
   145	            if any(isinstance(c, ast.Constant) and isinstance(c.value, str) for c in node.comparators):
   146	                offenders.append(f"{py_file.name}:{node.lineno}: {ast.unparse(node)}")
   147	
   148	    assert offenders == [], (

   363	def make_executor(session: ClaimSession) -> Executor:
   364	    """Return a ToolExecutor callable bound to this session."""
   365	
   366	    def execute(name: str, tool_input: dict[str, Any]) -> str:
   367	        handler = _DISPATCH.get(name)
   368	        if handler is None:
   369	            return _err("permanent", False, f"unknown tool: {name}")
   370	        try:
   371	            return handler(session, tool_input)
   372	        except Exception as exc:
   373	            # Defensive: anything raised inside a handler becomes a graceful error,
   374	            # never crashes the loop.
   375	            return _err("transient", True, f"{type(exc).__name__}: {exc}")

     1	# E-Commerce Platform — Shared Claude Code Conventions
     2	
     3	This file is the **project-level** entry point for the e-commerce monorepo. Every teammate's Claude session loads it automatically.
     4	
     5	## Scope: what belongs here vs. elsewhere
     6	
     7	Claude Code resolves configuration in a hierarchy:
     8	
     9	| Scope | Path | Purpose |
    10	|-------|------|---------|
    11	| **Project-level** | `./CLAUDE.md`, `.claude/standards/`, `.claude/rules/` | Conventions for *this* repo. Lives in git. Shared with the whole team. |
    12	| **User-level** | `~/.claude/CLAUDE.md`, `~/.claude/commands/`, `~/.claude/skills/` | Personal preferences. **Not shared via version control** — they stay on your laptop and never reach teammates. |
    13	| **Directory-level** | `path/CLAUDE.md` inside a subdir | Conventions narrower than the whole repo. We prefer `.claude/rules/` with glob `paths:` instead, so cross-cutting conventions (e.g. test files everywhere) work cleanly. |
    14	
    15	If something would only matter to you — your preferred commit-message style, your editor-specific snippets, a personal `/morning` summary command — put it under `~/.claude/`. Anything the whole team should agree on goes here in `./CLAUDE.md` or under `.claude/standards/`.
    16	
    17	## Shared standards (modular via @-imports)
    18	
    19	The actual conventions live in focused files so this entry point stays scannable:
    20	
    21	- @.claude/standards/frontend.md
    22	- @.claude/standards/api.md
    23	- @.claude/standards/database.md
    24	- @.claude/standards/testing.md
    25	
    26	Path-scoped rules in [.claude/rules/](.claude/rules/) layer on top of these standards and activate only when Claude is editing matching files (React components, API handlers, test files).
    27	
    28	## Repository layout
    29	
    30	```
    31	src/components/   React components (functional + hooks)
    32	src/pages/        Top-level routes
    33	src/api/          Node.js API handlers
    34	src/db/           PostgreSQL repository-pattern modules
    35	```
    36	
    37	Tests are co-located: `Foo.tsx` lives next to `Foo.test.tsx`.
    38

    40	
    41	**Convention isn't being followed?** Run `/memory` in your Claude Code session to see exactly which configuration files actually loaded. The most common cause is an instruction that lives in `~/.claude/CLAUDE.md` (user-level, never shared) when it should be in this file or under `.claude/standards/`.

    21	class Scratchpad:
    22	    def __init__(self, path: Path) -> None:
    23	        self.path = path
    24	
    25	    def append(self, entry: ScratchpadEntry) -> None:
    26	        self.path.parent.mkdir(parents=True, exist_ok=True)
    27	        line = entry.model_dump_json() + "\n"
    28	        with open(self.path, "ab") as f:
    29	            f.write(line.encode("utf-8"))
    30	            f.flush()
    31	            os.fsync(f.fileno())
    32	
    33	    def read(self) -> list[ScratchpadEntry]:
    34	        if not self.path.exists():
    35	            return []
    36	        entries: list[ScratchpadEntry] = []
    37	        with open(self.path, "rb") as f:
    38	            for raw in f:
    39	                line = raw.decode("utf-8").strip()
    40	                if line:
    41	                    entries.append(ScratchpadEntry.model_validate_json(line))
    42	        return entries
