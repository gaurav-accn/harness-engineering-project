     1	"""The agentic loop.
     2	
     3	The defining contract:
     4	- Control flow is driven by `response.stop_reason`.
     5	- Loop continues iff stop_reason == "tool_use".
     6	- Loop returns iff stop_reason == "end_turn".
     7	- Any other stop_reason raises UnexpectedStopReason.
     8	
     9	No string-membership tests against assistant text drive control flow here.
    10	No integer-literal iteration cap is the primary stopping mechanism.
    11	A Budget (token + wall-clock, sourced from config) is the safety net and raises
    12	BudgetExceeded if hit — it never silently truncates.
    13	"""
    14	
    15	import time
    16	from collections.abc import Callable
    17	from dataclasses import dataclass, field
    18	from typing import Any, Protocol
    19	
    20	from claims_intake.budget import Budget
    21	from claims_intake.tracer import Tracer
    22	
    23	
    24	class UnexpectedStopReason(Exception):
    25	    pass
    26	
    27	
    28	ToolExecutor = Callable[[str, dict[str, Any]], str]
    29	"""Takes (tool_name, tool_input) and returns the tool_result content as a string.
    30	Errors are returned as a serialized JSON string with is_error=true; never raised."""
    31	
    32	
    33	class MessagesClient(Protocol):
    34	    def create(self, **kwargs: Any) -> Any: ...
    35	
    36	
    37	@dataclass
    38	class FinalState:
    39	    messages: list[dict[str, Any]]
    40	    total_input_tokens: int
    41	    total_output_tokens: int
    42	    turn_count: int
    43	    final_content: list[Any] = field(default_factory=list)
    44	
    45	
    46	def run(
    47	    *,
    48	    client: Any,
    49	    model: str,
    50	    system: str,
    51	    tools: list[dict[str, Any]],
    52	    messages: list[dict[str, Any]],
    53	    tool_executor: ToolExecutor,
    54	    budget: Budget,
    55	    tracer: Tracer,
    56	    max_tokens: int = 4096,
    57	) -> FinalState:
    58	    """Run the agentic loop until the model returns stop_reason == "end_turn".
    59	
    60	    The loop condition is `response.stop_reason == "tool_use"` — nothing else.
    61	    There is no `while turn < N` cap. The Budget object is the safety mechanism.
    62	    """
    63	    working_messages = list(messages)
    64	    turn = 0
    65	    total_input = 0
    66	    total_output = 0
    67	
    68	    while True:
    69	        turn += 1
    70	        budget.check()
    71	        t0 = time.monotonic()
    72	        response = client.messages.create(
    73	            model=model,
    74	            max_tokens=max_tokens,
    75	            system=system,
    76	            tools=tools,
    77	            messages=working_messages,
    78	        )
    79	        latency_ms = (time.monotonic() - t0) * 1000.0
    80	
    81	        input_tokens = int(response.usage.input_tokens)
    82	        output_tokens = int(response.usage.output_tokens)
    83	        total_input += input_tokens
    84	        total_output += output_tokens
    85	        budget.record_input_tokens(input_tokens)
    86	
    87	        tool_calls = [
    88	            {"id": b.id, "name": b.name, "input": b.input}
    89	            for b in response.content
    90	            if getattr(b, "type", None) == "tool_use"
    91	        ]
    92	        tracer.write(
    93	            {
    94	                "turn": turn,
    95	                "stop_reason": response.stop_reason,
    96	                "tool_calls": tool_calls,
    97	                "latency_ms": round(latency_ms, 1),
    98	                "input_tokens": input_tokens,
    99	                "output_tokens": output_tokens,
   100	            }
   101	        )
   102	
   103	        if response.stop_reason == "end_turn":
   104	            working_messages.append({"role": "assistant", "content": response.content})
   105	            return FinalState(
   106	                messages=working_messages,
   107	                total_input_tokens=total_input,
   108	                total_output_tokens=total_output,
   109	                turn_count=turn,
   110	                final_content=list(response.content),
   111	            )
   112	
   113	        if response.stop_reason == "tool_use":
   114	            working_messages.append({"role": "assistant", "content": response.content})
   115	            tool_results: list[dict[str, Any]] = []
   116	            for block in response.content:
   117	                if getattr(block, "type", None) != "tool_use":
   118	                    continue
   119	                result_content = tool_executor(block.name, dict(block.input))
   120	                tool_results.append(
   121	                    {
   122	                        "type": "tool_result",
   123	                        "tool_use_id": block.id,
   124	                        "content": result_content,
   125	                    }
   126	                )
   127	            working_messages.append({"role": "user", "content": tool_results})
   128	            continue
   129	
   130	        raise UnexpectedStopReason(
   131	            f"turn {turn}: unexpected stop_reason={response.stop_reason!r}"
   132	        )

     1	"""Position-aware context assembly.
     2	
     3	Layout (top → bottom):
     4	  # Case Facts                       — top boundary, structured (≤ 600 tokens)
     5	  # Resolved: Refund inquiry         — middle, compressible zone (≤ 500 tokens)
     6	  # Resolved: Subscription cancellation — middle, compressible zone (≤ 500 tokens)
     7	  # Active issue: Payment-method update — bottom boundary, byte-exact verbatim
     8	
     9	This places key findings at both context boundaries (Case Facts at top, the
    10	active turn-by-turn at bottom against the new user turn) and lets the resolved
    11	narrative occupy the lower-attention middle. Sections are exclusive — no
    12	interleaving.
    13	"""
    14	from __future__ import annotations
    15	
    16	from dataclasses import dataclass
    17	
    18	from retail_context.case_facts import CaseFacts
    19	from retail_context.compressor import Compressed
    20	from retail_context.tokens import count
    21	
    22	# Display titles for the resolved sections. Exact strings are part of the
    23	# contract; the assemble-order audit regex matches against these.
    24	RESOLVED_TITLES: dict[str, str] = {
    25	    "refund": "# Resolved: Refund inquiry",
    26	    "subscription": "# Resolved: Subscription cancellation",
    27	}
    28	ACTIVE_TITLES: dict[str, str] = {
    29	    "payment_update": "# Active issue: Payment-method update",
    30	}
    31	
    32	
    33	@dataclass
    34	class AssembledContext:
    35	    markdown: str
    36	    case_facts_block: str
    37	    resolved_blocks: dict[str, str]
    38	    active_block: str
    39	    active_raw_text: str  # byte-exact verbatim source for the active segment
    40	
    41	    def section_tokens(self) -> dict[str, int]:
    42	        sections = {"case_facts": count(self.case_facts_block)}
    43	        for issue_id, block in self.resolved_blocks.items():
    44	            sections[f"resolved_{issue_id}"] = count(block)
    45	        sections["active"] = count(self.active_block)
    46	        return sections
    47	
    48	    def total_tokens(self) -> int:
    49	        return count(self.markdown)
    50	
    51	
    52	def build(case_facts: CaseFacts, compressed: Compressed) -> AssembledContext:
    53	    case_block = case_facts.to_markdown().rstrip() + "\n"
    54	
    55	    resolved_blocks: dict[str, str] = {}
    56	    # Preserve declaration order: refund then subscription
    57	    for issue_id in ("refund", "subscription"):
    58	        if issue_id not in compressed.summaries:
    59	            raise KeyError(f"compressed summaries missing required issue {issue_id!r}")
    60	        title = RESOLVED_TITLES[issue_id]
    61	        body = compressed.summaries[issue_id].text.strip()
    62	        resolved_blocks[issue_id] = f"{title}\n\n{body}\n"
    63	
    64	    active_title = ACTIVE_TITLES.get(
    65	        compressed.active_issue_id,
    66	        f"# Active issue: {compressed.active_issue_id}",
    67	    )
    68	    active_block = f"{active_title}\n\n{compressed.active_text}\n"
    69	
    70	    markdown = (
    71	        case_block
    72	        + "\n"
    73	        + resolved_blocks["refund"]
    74	        + "\n"
    75	        + resolved_blocks["subscription"]
    76	        + "\n"
    77	        + active_block
    78	    )
    79	
    80	    return AssembledContext(
    81	        markdown=markdown,
    82	        case_facts_block=case_block,
    83	        resolved_blocks=resolved_blocks,
    84	        active_block=active_block,
    85	        active_raw_text=compressed.active_text,
    86	    )

     1	"""Eval-question runner.
     2	
     3	Issues one Claude call per question with the assembled context as the system prompt
     4	and the question as the user turn. A question passes when the expected fragment
     5	appears (case-insensitive) in the model's answer. Pass threshold ≥ 5/6.
     6	
     7	Also runs a control variant with the Case Facts block removed; Q1 (refund amount)
     8	and Q6 (exact status token) are the control questions; at least one must FAIL in the
     9	control — that's how we verify the
    10	case-facts block is doing real work rather than being coincidentally redundant
    11	with the active verbatim section.
    12	"""
    13	from __future__ import annotations
    14	
    15	import json
    16	from dataclasses import dataclass
    17	from pathlib import Path
    18	from typing import Any
    19	
    20	from retail_context.client import complete_with_system
    21	
    22	EVAL_SYSTEM_PREFIX = (
    23	    "You are a retail customer-support assistant for Pantry Plus. The conversation"
    24	    " context for the current customer is provided below. Answer the question concisely"
    25	    " using only information present in the context. If the answer is not present,"
    26	    " say 'unknown' — do not invent.\n\n"
    27	)
    28	
    29	
    30	@dataclass
    31	class EvalResult:
    32	    question_id: str
    33	    question: str
    34	    expected_fragment: str
    35	    model_answer: str
    36	    passed: bool
    37	    input_tokens: int
    38	    output_tokens: int
    39	
    40	    def to_dict(self) -> dict[str, Any]:
    41	        return {
    42	            "question_id": self.question_id,
    43	            "question": self.question,
    44	            "expected_fragment": self.expected_fragment,
    45	            "model_answer": self.model_answer,
    46	            "passed": self.passed,
    47	            "input_tokens": self.input_tokens,
    48	            "output_tokens": self.output_tokens,
    49	        }
    50	
    51	
    52	def _passed(expected: str, answer: str) -> bool:
    53	    return expected.lower() in answer.lower()
    54	
    55	
    56	def run_questions(
    57	    questions: list[dict[str, Any]],
    58	    context_markdown: str,
    59	    *,
    60	    model: str | None = None,
    61	) -> list[EvalResult]:
    62	    system = EVAL_SYSTEM_PREFIX + context_markdown
    63	    results: list[EvalResult] = []
    64	    for q in questions:
    65	        answer, in_tok, out_tok = complete_with_system(
    66	            system,
    67	            q["question"],
    68	            model=model,
    69	            max_tokens=256,
    70	        )
    71	        results.append(
    72	            EvalResult(
    73	                question_id=q["id"],
    74	                question=q["question"],
    75	                expected_fragment=q["expected_fragment"],
    76	                model_answer=answer,
    77	                passed=_passed(q["expected_fragment"], answer),
    78	                input_tokens=in_tok,
    79	                output_tokens=out_tok,
    80	            )
    81	        )
    82	    return results
    83	
    84	
    85	def strip_case_facts(context_markdown: str) -> str:
    86	    """Return the assembled context with the `# Case Facts` block removed.
    87	
    88	    The control variant — at least one of Q1 / Q6 should fail when this block is gone.

    22	    if not state.steps:
    23	        return "fresh"
